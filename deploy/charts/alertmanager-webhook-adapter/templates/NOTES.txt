1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ include "awa.namespace" . }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "awa.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ include "awa.namespace" . }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
        You can watch its status by running:
          kubectl get --namespace {{ include "awa.namespace" . }} svc -w {{ include "awa.fullname" . }}
  export SERVICE_IP=$(kubectl get svc --namespace {{ include "awa.namespace" . }} {{ include "awa.fullname" . }} --template "{{ "{{ range (index .status.loadBalancer.ingress 0) }}{{ . }}{{ end }}" }}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ include "awa.namespace" . }} -l "app.kubernetes.io/name={{ include "awa.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ include "awa.namespace" . }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8080/webhook/send?channel_type=weixin&token=YOUR_TOKEN to use your application"
  kubectl --namespace {{ include "awa.namespace" . }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}

2. Configure Alertmanager webhook URL:
   http://{{ include "awa.fullname" . }}.{{ include "awa.namespace" . }}.svc/webhook/send?channel_type=weixin&token=YOUR_TOKEN
